SQL idea

WITH paid AS (
  SELECT invoice_id, SUM(amount_cents) AS paid_cents
  FROM payments GROUP BY invoice_id
)
SELECT invoices.total_cents - COALESCE(paid.paid_cents, 0) AS balance_cents
FROM invoices LEFT JOIN paid USING (invoice_id)
WHERE invoices.due_date < :cutoff
  AND balance_cents > 0;
